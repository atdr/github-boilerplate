# Pinned rather than floating: a formatter that changes its output in a patch
# release turns unrelated pull requests red.
mdformat==1.0.0
# Tables. Core CommonMark has none, and without this every padded table is
# reflowed into unpadded pipes.
mdformat-gfm==1.0.0
# YAML frontmatter, which skills in .claude/ carry. Without this mdformat
# rewrites it into a horizontal rule and a heading. Never add a codeformatter
# plugin (mdformat-black and the like): mdformat enables every installed one
# and would reformat code inside fences.
mdformat-frontmatter==2.1.2
# Footnotes, which GitHub renders. Without this mdformat escapes each
# definition (`\[^note\]:`), and the reference no longer links to it.
mdformat-footnote==0.1.3
